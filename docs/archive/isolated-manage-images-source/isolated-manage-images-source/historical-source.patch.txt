+def compatibility_identities(asset):
+    """Return the authenticated full-contract and compiled-projection identities."""
+    if not isinstance(asset, dict) or not isinstance(asset.get('compatibility'), dict):
+        raise ValueError('signed release compatibility contract is missing')

+        raise ValueError('signed release compatibility contract has unexpected fields')
+    runtime = contract['workstation_runtime']
+    if not isinstance(runtime, dict) or any(field not in runtime for field in PROJECTION_FIELDS):
+        raise ValueError('signed release compatibility runtime projection is incomplete')
+    projection = {

+        'manage': contract['manage'],
