def durable(sequence, active, complete):
    return {'schema': 'cybex.james.provisioning-state.v1', 'session_id': SESSION,
            'plan': copy.deepcopy(PLAN_BODY), 'manage_origin': MANAGE_ORIGIN,
            'management_signing_public_key_b64': MANAGEMENT_KEY,

            'device_public_key_fingerprint': KEY, 'next_event_sequence': sequence,
            'identity_active': active, 'installation_complete': complete}


def observed(sequence, active, complete, *, release=False):
    return {'durable': durable(sequence, active, complete),
            'durable_sha256': str(sequence) * 64,
            'partition': {'label': 'CYBEX_STATE', 'partuuid': 'part',

                          'last_sector': 33556479, 'sector_size': 512},
            'snapshot': {'sha256': 'd' * 64, 'size_bytes': 16 * q04.GIB,
                         'journal_replay': 'e2fsck-journal-only'},
            'appliance_release': {'sha256': 'a' * 64} if release else None}


def progress():
    return [{'sequence': sequence, 'stage': stage, 'status': status,
             'event_id': q04.deterministic_event_id(SESSION, PLAN, sequence)}
            for sequence, stage, status in q04.EVENTS]


class VM:
    def __init__(self, disk, iso, code, ovmf):
        self.running = True
        self.reset = False
        self.reset_observed = False
        self.quit_count = 0
        self.disk, self.iso, self.code, self.ovmf = disk, iso, code, ovmf


class Monitor:
    def __init__(self, vm):
        self.vm = vm
        self.socket = types.SimpleNamespace(close=lambda: None)

    def call(self, command, arguments=None):
        if command == 'query-status':
            return {'running': self.vm.running}
        if command == 'query-version':
            return {'qemu': {'major': 10, 'minor': 0, 'micro': 13}}
        if command == 'stop':
            self.vm.running = False; return {}
        if command == 'cont':
            self.vm.running = True
            self.vm.reset = self.vm.quit_count == 3
            return {}
        if command == 'quit':
            self.vm.running = False; self.vm.quit_count += 1; return {}
        if command == 'query-block':
            return [
                {'device': 'system', 'inserted': {'file': str(self.vm.disk), 'ro': False,
                                                  'image': {'filename': str(self.vm.disk)}},
                 'io-status': 'ok'},
                {'device': 'ide1-cd0', 'inserted': {'file': str(self.vm.iso), 'ro': True,
                                                    'image': {'filename': str(self.vm.iso)}}},
                {'device': 'pflash0', 'inserted': {'file': str(self.vm.code), 'ro': True,
                                                  'image': {'filename': str(self.vm.code)}}},
                {'device': 'pflash1', 'inserted': {'file': str(self.vm.ovmf), 'ro': False,
                                                  'image': {'filename': str(self.vm.ovmf)}}},
            ]
        raise AssertionError(command)

    def events(self):
        if self.vm.reset:
            self.vm.reset = False
            self.vm.reset_observed = True
            return [{'event': 'RESET', 'data': {'guest': True, 'reason': 'guest-reset'}}]
        return []


class InstallationPowerCutTests(unittest.TestCase):
    def inspector_fixture(self, root, *, fail_unmount=False):
        calls = []
        def run(arguments, **_kwargs):
            calls.append(arguments)
            command = arguments[0]
            if command == 'losetup':
                if '--detach' in arguments:
                    return subprocess.CompletedProcess(arguments, 0, '', '')
                loop = '/dev/loop9\n' if '--read-only' in arguments else '/dev/loop10\n'
                return subprocess.CompletedProcess(arguments, 0, loop, '')
            if command == 'blockdev':
                output = '512\n' if '--getss' in arguments else str(16 * q04.GIB) + '\n'
                return subprocess.CompletedProcess(arguments, 0, output, '')
            outputs = {
                'sgdisk': "First sector: 2048\nLast sector: 33556479\n"
                          "Partition GUID code: 8300 (Linux filesystem)\n"
                          "Partition name: 'CYBEX_STATE'\n",
                'udevadm': '',
                'lsblk': json.dumps({'blockdevices': [{'path': '/dev/loop9p1', 'type': 'part',
                    'partlabel': 'CYBEX_STATE', 'partuuid': 'part', 'fstype': 'ext4',
                    'uuid': 'filesystem', 'ro': True}]}),
                'e2fsck': '', 'mount': '',
                'findmnt': '/dev/loop10 ro,norecovery,nodev,nosuid,noexec\n',
                'umount': '',
            }
            return subprocess.CompletedProcess(arguments, 1 if command == 'umount' and fail_unmount else 0,
                                               outputs[command], '')
        def copy_snapshot(_source, destination, size):
            Path(destination).write_bytes(b'private-snapshot')
            return {'sha256': 'd' * 64, 'size_bytes': size, 'source_device': 9}
        return q04.ReadOnlyStateInspector(root, run=run, copy=copy_snapshot), calls

    def fixture(self, root, inspections):

        vm = VM(disk, iso, code, ovmf)
        state = {'ready': False}
        def api(path):
            if path.startswith('/v1/james/nodes/'):
                return {'node': {'device_id': DEVICE,
                    'james_reported_at': '2026-09-21T00:01:00Z' if state['ready'] else None,
                    'appliance_base_os': 'nixos', 'appliance_release': '0.2.1-dev.26',

                    'ready_at': '2026-09-21T00:01:00Z' if state['ready'] else None,
                    'install_plan': copy.deepcopy(PLAN_BODY),
                    'progress': progress() if state['ready'] else []}
        def inspect(_disk):
            return copy.deepcopy(inspections.pop(0))
        arguments = ['qemu-system-x86_64', '-uuid', HARDWARE['uuid'],
                     '-drive', f'if=pflash,format=raw,unit=0,readonly=on,file={code}',
                     '-drive', f'if=pflash,format=raw,unit=1,file={ovmf}',

            process_start=lambda pid: pid * 10,
            wait_exit=lambda _pid, _start: '2026-09-21T00:00:00Z',
            sleep=lambda _seconds: state.update(ready=vm.reset_observed),

    def test_three_exact_cuts_resume_same_identity_and_finish_after_reset(self):
        inspections = [observed(3, False, False), observed(3, False, False),
                       observed(5, True, False), observed(5, True, False),
                       observed(9, True, True, release=True),
                       observed(9, True, True, release=True)]
        with tempfile.TemporaryDirectory() as temporary:
            controller, vm, state = self.fixture(Path(temporary), inspections)
            controller.initialize(100)
            controller.cut('state_creation', 100)
            vm.running = False
            controller.resume('state_creation', 101)
            controller.cut('identity_activation', 101)
            vm.running = False
            controller.resume('identity_activation', 102)
            controller.cut('install_completion', 102)
            vm.running = False
            evidence = controller.finish(103)
        self.assertTrue(evidence['ok'])
        self.assertEqual([value['boundary'] for value in evidence['cuts']], list(q04.BOUNDARIES))
        self.assertEqual(vm.quit_count, 3)
        self.assertTrue(evidence['guest_initiated_reset_observed'])
        self.assertTrue(evidence['iso_attached_through_installed_ready'])
        self.assertTrue(evidence['visible_event_sequence_exact'])
        self.assertFalse(evidence['deterministic_event_ids_observed'])
        self.assertEqual(evidence['intentional_host_poweroffs'], 3)
        self.assertTrue(all(cut['host_poweroff']['process_exited_at'] for cut in evidence['cuts']))
        self.assertEqual(inspections, [])

    def test_qemu_and_qmp_bind_exact_hardware_disk_iso_and_firmware(self):
        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary)
            controller, vm, _state = self.fixture(root, [])
            arguments = controller.argv(100)
            identity = q04.qemu_identity(arguments)
            self.assertEqual(identity, {**HARDWARE, 'disk': str(vm.disk), 'iso': str(vm.iso),
                                        'ovmf_code': str(root / 'OVMF_CODE.fd'),
                                        'ovmf_vars': str(root / 'OVMF_VARS.fd')})
            pins = controller._pins()
            (root / 'OVMF_CODE.fd').write_bytes(b'CODE')
            self.assertNotEqual(controller._pins(), pins)
            (root / 'OVMF_CODE.fd').write_bytes(b'code')
            blocks = Monitor(vm).call('query-block')
            self.assertTrue(q04.verify_qmp_attachments(
                blocks, vm.disk, vm.iso, root / 'OVMF_CODE.fd', root / 'OVMF_VARS.fd'))
            blocks[1]['inserted']['ro'] = False
            with self.assertRaisesRegex(ValueError, 'not read-only'):
                q04.verify_qmp_attachments(
                    blocks, vm.disk, vm.iso, root / 'OVMF_CODE.fd', root / 'OVMF_VARS.fd')
            blocks[1]['inserted']['ro'] = True
            blocks[2]['inserted']['ro'] = False
            with self.assertRaisesRegex(ValueError, 'OVMF CODE'):
                q04.verify_qmp_attachments(
                    blocks, vm.disk, vm.iso, root / 'OVMF_CODE.fd', root / 'OVMF_VARS.fd')

    def test_qmp_socket_is_bound_to_the_supplied_qemu_pid(self):
        class Peer:
            def __init__(self, pid):
                self.pid = pid
            def getsockopt(self, level, option, size):
                self_args = (level, option, size)
                if self_args != (q04.socket.SOL_SOCKET, q04.socket.SO_PEERCRED,
                                  q04.struct.calcsize('3i')):
                    raise AssertionError(self_args)
                return q04.struct.pack('3i', self.pid, 1000, 1000)
        self.assertTrue(q04.verify_qmp_peer(Peer(123), 123))
        with self.assertRaisesRegex(ValueError, 'supplied QEMU PID'):
            q04.verify_qmp_peer(Peer(124), 123)

    def test_wait_process_exit_accepts_the_exact_unreaped_zombie(self):
        process = subprocess.Popen(['sleep', '60'])
        try:
            starttime = q04.process_starttime(process.pid)
            process.terminate()
            self.assertTrue(q04.wait_process_exit(
                process.pid, starttime, timeout=2, interval=.01))
            self.assertEqual(q04.process_state_starttime(process.pid), ('Z', starttime))
        finally:
            process.wait()

    def test_durable_state_path_matches_installer_layout_transition(self):
        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary)
            flat = root / 'provisioning-state.json'
            flat.write_text('{}')
            self.assertEqual(q04.durable_state_path(root), flat)
            protected = root / 'control/provisioning-state.json'
            protected.parent.mkdir()

    def test_event_replay_requires_one_exact_deterministic_sequence(self):

    def test_read_only_inspector_always_detaches_partial_state_attachment(self):
        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary); disk = root / 'disk.raw'; disk.write_bytes(b'disk')
            inspector, calls = self.inspector_fixture(root)
            real_euid = q04.os.geteuid()
            with mock.patch.object(q04.os, 'geteuid', side_effect=[0] + [real_euid] * 8):
                with self.assertRaises(q04.StateNotReady):
                    inspector(disk)
        self.assertIn(['mount', '-t', 'ext4', '-o', 'ro,noload,nodev,nosuid,noexec',
                       '/dev/loop10', str(root / 'q04-state-inspection')], calls)
        self.assertIn(['e2fsck', '-E', 'journal_only', '-p', '/dev/loop10'], calls)
        self.assertEqual(calls[-2][0], 'umount')
        self.assertEqual(calls[-1], ['losetup', '--detach', '/dev/loop10'])

    def test_read_only_inspector_detaches_even_when_unmount_fails(self):
        with tempfile.TemporaryDirectory() as temporary:
            root = Path(temporary); disk = root / 'disk.raw'; disk.write_bytes(b'disk')
            inspector, calls = self.inspector_fixture(root, fail_unmount=True)
            real_euid = q04.os.geteuid()
            with mock.patch.object(q04.os, 'geteuid', side_effect=[0] + [real_euid] * 8):
                with self.assertRaisesRegex(ValueError, 'residual private STATE resources'):
                    inspector(disk)
            self.assertEqual(len(list(root.glob('q04-state-snapshot.*.ext4'))), 1)
            self.assertEqual(calls[-1][0], 'umount')
            self.assertNotIn(['losetup', '--detach', '/dev/loop10'], calls)


class Clock:
    def __init__(self):
        self.value = 0
    def __call__(self):
        self.value += .01
        return self.value
